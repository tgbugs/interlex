* ops
See [[*core flow][core flow]] or copy this ~config.example.org~ file to ~config.org~
and edit the variables sections and tangle with ~C-c C-v C-t~ or run.

The resulting ~${config_profile}/config~ folder should be copied to the profile host
along with ~${config_profile}/workflow-funs-ilx.sh~.

To change which configuration will tangle add or remove ~COMMENT~ on a section.
Only one section can be active at a time. If multiple sections are active the
exact value for a variable is usually the one that appears first in the file.

The setup process expects to run on two systems, one orchestrator (orch) and
the remote profile host (host). The orchestrator needs gnu coreutils, bash,
docker, podman, emacs, cat, and tar. The host needs gnu coreutils, bash, docker,
cat, and tar.

function cleanup() {
command -v docker && {
  running=$(docker ps -qf ancestor=tgbugs/musl:interlex)
  for cid in ${running}; do
    docker stop --timeout 10 ${cid};
  done
  }
}
trap cleanup SIGTERM SIGINT
sleep infinity &
wait $!

**** cgroups
On some systems in order to run docker inside a rootless podman
container there are some additional steps that are required.
You may be able to find other ways to obtain a user writable
=/sys/fs/cgroup= owned by root instead of nobody in the container.
Note that I do not fully understand the security implications of
doing this. However it seems safer than running podman as root.

src_elisp[:results raw]{(user-login-name)}

#+name: &podman-user
: user

#+begin_src bash :dir /su:: :noweb yes :eval never 
USER_UID=$(id -u <<&podman-user()>>)
USER_CGROUP=/sys/fs/cgroup/user.slice/user-${USER_UID}.slice

# clean up session if it exists
rmdir "${USER_CGROUP}/session" 2>/dev/null || true

# enable parent controllers
echo "+cpu +memory +pids +io" > /sys/fs/cgroup/cgroup.subtree_control
mkdir -p /sys/fs/cgroup/user.slice
echo "+cpu +memory +pids +io" > /sys/fs/cgroup/user.slice/cgroup.subtree_control

# user cgroup directory
mkdir -p ${USER_CGROUP}

# enable user cgroup controller before session
echo "+cpu +memory +pids +io" > ${USER_CGROUP}/cgroup.subtree_control

# create session
mkdir -p ${USER_CGROUP}/session

# chow the user cgroup
chown -R <<&podman-user()>>:<<&podman-user()>> ${USER_CGROUP}
#+end_src

When you need to run docker inside podman start podman in a shell with the following.
#+begin_src bash :dir /su:: :noweb yes :eval never
USER_UID=$(id -u <<&podman-user()>>)

# as root move current process into the session
echo $$ > /sys/fs/cgroup/user.slice/user-${USER_UID}.slice/session/cgroup.procs

# drop in as the podman user to retain some of the advantages of rootless
su - <<&podman-user()>>
#+end_src

*** server :host:
A point about profile host =/var/lib/interlex= vs the container =/var/lib/interlex=.
We use the host =/var/lib/interlex= as a standard location for saving configuration
files however it should NEVER be mounted to the container directly.

#+name: &run-ilx-server
#+begin_src bash :noweb yes
function docker-run-ilx-server () {
local entrypoint cfg dhp lessl
entrypoint=/etc/entrypoints/${FUNCNAME##*-}.sh
cfg=/var/lib/interlex/config
dhp=/etc/ssl/certs/dhparam.pem
lessl=/etc/letsencrypt/live/<<&profile-hostname()>>
[ -d ${cfg} ]   || { echo "missing config directory ${cfg}" ; return 1; }
[ -d ${lessl} ] || { echo "missing ssl certs ${lessl}"; }  # can't error here because we need to be able to test w/ self signed
[ -f ${dhp} ]   || { echo "missing ${dhp} run openssl dhparam -out /tmp/dhparam.pem 4096 and scp it over" ; return 3; }

[ -d /tmp/ilx-scratch ] || mkdir /tmp/ilx-scratch

chmod 0770 ${cfg}

_cid_new=$(
docker run \
-d \
--name ilx-server \
--restart unless-stopped \
${_entrypoints:+-v ${_entrypoints}:/etc/entrypoints} \
-v ${cfg}:/root/ilx-config \
-v ${dhp}:/etc/ssl/certs/dhparam.pem \
-v /etc/letsencrypt:/etc/letsencrypt \
-v /var/log/interlex:/var/log/interlex \
-v /var/log/nginx:/var/log/nginx \
-v /var/log/letsencrypt:/var/log/letsencrypt \
-v /tmp/ilx-scratch/ontologies:/tmp/ontologies \
-v /tmp/ilx-scratch/interlex-load:/tmp/interlex-load \
--entrypoint ${entrypoint} \
-p 80:80 \
-p 443:443 \
tgbugs/musl:interlex
)
code=$?
[ ${code} = 0 ] || return ${code}
echo ilx server container ${_cid_new}
sleep 10
docker logs ${_cid_new}
docker exec ${_cid_new} sh -c 'rc-service interlex status && rc-service nginx status'
}
